/app/clue/.venv/bin/python -I -S -B - <<'PY'
import ast
import hashlib
import sys
from pathlib import Path

source = Path("/app/clue/src/clue/control.py").read_text(
    encoding="utf-8-sig"
)
tree = ast.dump(ast.parse(source), include_attributes=False)

print("PYTHON:", sys.version.split()[0])
print("UTF8_LF_SHA256:", hashlib.sha256(source.encode("utf-8")).hexdigest())
print("AST_SHA256:", hashlib.sha256(tree.encode("utf-8")).hexdigest())
PY
